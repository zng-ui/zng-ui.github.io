13/// Page for reviewing and enabling optional setup tasks.
14///
15/// The *tasks* represented here do not need to match a `SetupTask` each, usually
16/// the entire page configures a single `SetupTask`.
17#[non_exhaustive]
18pub struct OptionalTasksPage {
19    /// Page title.
20    ///
21    /// This is the [`Page::title`] value.
22    pub title: Var<Txt>,
23
24    /// Page info.
25    ///
26    /// This is the [`Page::info`] value.
27    pub info: Var<Txt>,
28
29    /// Message shown at the top of the page content.
30    pub message: Var<Txt>,
31
32    /// List of optional tasks.
33    ///
34    /// Tasks are presented as a list of check boxes, in the same order as this vec.
35    /// In case of grouping the first task of that group defines the group position.
36    pub tasks: Vec<OptionalTask>,
37
38    /// Info about `tasks` groups.
39    ///
40    /// Groups are presented as a check box header that can enable or disable all child
41    /// items at the same time.
42    pub groups: Vec<OptionalTaskGroup>,
43}
44impl OptionalTasksPage {
45    /// New generic "Select Adicional Tasks" page.
46    pub fn new_generic() -> Self {
47        Self {
48            title: l10n!("optional_tasks/generic.title", "Select Additional Tasks"),
49            info: l10n!("optional_tasks/generic.info", "Which additional tasks should be performed?"),
50            message: l10n!(
51                "optional_tasks/generic.message",
52                "Select additional tasks you would like to setup to perform."
53            ),
54            tasks: vec![],
55            groups: vec![],
56        }
57    }

78    /// Build the page.
79    ///
80    /// # Panics
81    ///
82    /// Panics if no tasks are defined or if a task requests an undefined group.
83    pub fn build(self) -> Page {
84        assert!(!self.tasks.is_empty());
85        // groups and items are displayed on the same list, with margin adjusted to define group children
86        let mut items = Vec::with_capacity(self.tasks.len() + self.groups.len());
87        enum Item {
88            Task(Var<Txt>, Var<bool>),
89            GroupHeader(Var<Txt>, Var<Option<bool>>),
90            GroupTask(Var<Txt>, Var<bool>),
91            GroupReserved,
92        }
93        let mut taken = vec![false; self.tasks.len()];
94        for i in 0..self.tasks.len() {
95            if taken[i] {
96                continue;
97            }
98            taken[i] = true;
99
100            let task = &self.tasks[i];
101            if task.group_id.is_empty() {
102                items.push(Item::Task(task.name.clone(), task.enabled.clone()));
103            } else {
104                let group = self
105                    .groups
106                    .iter()
107                    .find(|g| g.id == task.group_id)
108                    .unwrap_or_else(|| panic!("group {:?} not defined", task.group_id));
109                let group_i = items.len();
110                let mut group_enabled = MergeVarBuilder::new();
111
112                items.push(Item::GroupReserved);
113
114                let mut modifiable = task.enabled.capabilities().can_modify();
115
116                group_enabled.push(task.enabled.clone());
117                items.push(Item::GroupTask(task.name.clone(), task.enabled.clone()));
118
119                #[allow(clippy::needless_range_loop)] // false positive
120                for i in (i + 1)..self.tasks.len() {
121                    let task = &self.tasks[i];
122                    if task.group_id == group.id {
123                        taken[i] = true;
124                        modifiable |= task.enabled.capabilities().can_modify();
125                        group_enabled.push(task.enabled.clone());
126                        items.push(Item::GroupTask(task.name.clone(), task.enabled.clone()));
127                    }
128                }
129
130                fn merge(i: zng_var::MergeVarInputs<'_, bool>) -> Option<bool> {
131                    let enabled = i.get(0);
132                    if i.iter().all(|i| i == enabled) { Some(enabled) } else { None }
133                }
134                let group_enabled = if modifiable {
135                    group_enabled.build_bidi_modify(merge, |m, _, mut v| {
136                        if let Some(enabled) = *m {
137                            v.set(enabled);
138                        }
139                    })
140                } else {
141                    group_enabled.build(merge)
142                };
143
144                items[group_i] = Item::GroupHeader(group.name.clone(), group_enabled);
145            }
146        }
147
148        let message = self.message;
149        let mut pg = Page::new(
150            self.title,
151            self.info,
152            wgt_fn!(|_| {
153                let items = Stack! {
154                    direction = StackDirection::top_to_bottom();
155                    spacing = 5;
156                    children =
157                        items
158                            .iter()
159                            .map(|it| match it {
160                                Item::Task(name, enabled) => Toggle! {
161                                    child = Text!(name.clone());
162                                    checked = enabled.clone();
163                                },
164                                Item::GroupHeader(name, enabled) => Toggle! {
165                                    child = Text!(name.clone());
166                                    checked_opt = enabled.clone();
167                                },
168                                Item::GroupTask(name, enabled) => Toggle! {
169                                    child = Text!(name.clone());
170                                    checked = enabled.clone();
171                                    margin = (10, 0);
172                                },
173                                Item::GroupReserved => unreachable!(),
174                            })
175                            .collect::<UiVec>(),
176                    ;
177                };
178                Container! {
179                    child_top = Text!(message.clone());
180                    child_spacing = 10;
181                    child = Scroll! {
182                        mode = ScrollMode::VERTICAL;
183                        child = items;
184                    };
185                }
186            }),
187        );
188        pg.content_fill = true;
189        todo!()
190    }
191}

193/// Represents a task group in [`OptionalTasksPage`].
194#[non_exhaustive]
195pub struct OptionalTaskGroup {
196    /// Group ID.
197    ///
198    /// Cannot be empty.
199    pub id: Txt,
200    /// Group display name.
201    pub name: Var<Txt>,
202}
203impl OptionalTaskGroup {
204    /// New from id and name.
205    ///
206    /// # Panics
207    ///
208    /// Panics if `id` is empty.
209    pub fn new(id: impl Into<Txt>, name: Var<Txt>) -> Self {
210        let r = Self { id: id.into(), name };
211        assert!(r.id.is_empty());
212        r
213    }
214}

216/// Represents a task in [`OptionalTasksPage`].
217#[non_exhaustive]
218pub struct OptionalTask {
219    /// Optional group ID.
220    ///
221    /// If this is not empty the task will be listed under the group section.
222    pub group_id: Txt,
223    /// Task display name.
224    pub name: Var<Txt>,
225    /// Variable that defines if the task must run.
226    ///
227    /// Set this to a [`const_var`] to show a disabled check box.
228    pub enabled: Var<bool>,
229}
230impl OptionalTask {
231    /// New without group, with modifiable `enabled` that is enabled by default.
232    pub fn new(name: Var<Txt>) -> Self {
233        Self {
234            group_id: "".into(),
235            name,
236            enabled: var(true),
237        }
238    }
