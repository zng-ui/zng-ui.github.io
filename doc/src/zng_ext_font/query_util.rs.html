9struct FontiqueCache {
10    collection: fontique::Collection,
11    source_cache: fontique::SourceCache,
12    ipc_cache: Vec<(fontique::FamilyId, usize, WeakFontBytes)>,
13}

15app_local! {
16    static FONTIQUE_CACHE: Option<FontiqueCache> = const { None };
17}
18
19pub fn invalidate_fontique_cache() {
20    *FONTIQUE_CACHE.write() = None;
21}
22
23fn fontique_cache() -> MappedRecursiveRwLockWriteGuard<'static, FontiqueCache> {
24    let mut c = FONTIQUE_CACHE.write();
25    if c.is_none() {
26        let collection = fontique::Collection::new(fontique::CollectionOptions {
27            shared: false,
28            system_fonts: true,
29        });
30        let source_cache = fontique::SourceCache::new(fontique::SourceCacheOptions { shared: false });
31        *c = Some(FontiqueCache {
32            collection,
33            source_cache,
34            ipc_cache: vec![],
35        });
36    }
37    MappedRecursiveRwLockWriteGuard::map(c, |c| c.as_mut().unwrap())
38}

40pub fn system_all() -> ResponseVar<Vec<FontName>> {
41    zng_task::wait_respond(|| {
42        fontique_cache()
43            .collection
44            .family_names()
45            .map(|n| Txt::from_str(n).into())
46            .collect()
47    })
48}

50/// Returns (Some(bytes, collection_index, variable_cfg))
51pub fn best(
52    font_name: &FontName,
53    style: FontStyle,
54    weight: FontWeight,
55    stretch: FontStretch,
56) -> Result<Option<(FontBytes, u32, Vec<Variation>)>, FontLoadingError> {
57    use fontique::QueryFamily;
58
59    // fontique does basic replacement for QueryFamily::Generic,
60    // but generic names only appear here if they where not resolved by FONTS
61    // where we already define replacements
62    let family = if cfg!(target_os = "macos") && font_name == "system-ui" {
63        // macOS has a special function to get the UI font
64        QueryFamily::Generic(fontique::GenericFamily::SystemUi)
65    } else {
66        // on Linux fontconfig handles generic names so we can just pass the name
67        // other systems will just not match
68        QueryFamily::Named(font_name.name())
69    };
70
71    let attributes = fontique::Attributes {
72        width: fontique::FontWidth::from_ratio(stretch.0),
73        weight: fontique::FontWeight::new(weight.0),
74        style: match style {
75            FontStyle::Normal => fontique::FontStyle::Normal,
76            FontStyle::Italic => fontique::FontStyle::Italic,
77            FontStyle::Oblique => fontique::FontStyle::Oblique(None),
78        },
79    };
80
81    let mut c = fontique_cache();
82    let c = &mut *c;
83
84    let mut matched = None;
85    {
86        let mut q = c.collection.query(&mut c.source_cache);
87        q.set_families([family]);
88        q.set_attributes(attributes);
89        q.matches_with(|f| {
90            // render synthesis are inferred from the font (synthesis_for),
91            // variation config is not really synthesis so its collected here
92            let var_cfg: Vec<Variation> = f
93                .synthesis
94                .variation_settings()
95                .iter()
96                .map(|&(tag, value)| Variation {
97                    tag: Tag::from_be_bytes(tag.to_be_bytes()),
98                    value,
99                })
100                .collect();
101
102            matched = Some((f.family, f.index, var_cfg));
103            fontique::QueryStatus::Stop
104        });
105    }
106    if matched.is_none() {
107        tracing::debug!(target: "font_loading", "system font not found\nquery: {:?}", (font_name, style, weight, stretch));
108        return Ok(None);
109    }
110    let ((family_id, font_file_idx), font_collection_idx, variable_cfg) = matched.unwrap();
111
112    let mut active_ipc = None;
113    c.ipc_cache.retain(|(f, i, b)| {
114        if f == &family_id && *i == font_file_idx {
115            if let Some(b) = b.upgrade() {
116                active_ipc = Some(b);
117                true
118            } else {
119                false
120            }
121        } else {
122            b.strong_count() > 0
123        }
124    });
125    if let Some((_, _, b)) = c.ipc_cache.iter().find(|(f, i, _)| f == &family_id && *i == font_file_idx)
126        && let Some(b) = b.upgrade()
127    {
128        return Ok(Some((b, font_collection_idx, variable_cfg)));
129    }
130
131    let family_info = c.collection.family(family_id).unwrap();
132    let bytes = match &family_info.fonts()[font_file_idx].source().kind {
133        fontique::SourceKind::Memory(b) => FontBytes::from_ipc(IpcBytes::from_slice_blocking(b.as_ref())?),
134        fontique::SourceKind::Path(p) => FontBytes::from_file(p.to_path_buf())?,
135    };
136    c.ipc_cache.push((family_id, font_file_idx, bytes.downgrade()));
137
138    Ok(Some((bytes, font_collection_idx, variable_cfg)))
139}
