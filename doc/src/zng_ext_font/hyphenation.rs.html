3/// Hyphenation service.
4///
5/// Note that embedded data is required to support a language, if the feature `"hypher_all"` is enabled
6/// data for all supported languages is embedded (adds some 1.1MB of data), otherwise you must add a `hypher` dependency
7/// and select specific languages.
8///
9/// ```toml
10/// [dependencies]
11/// # only embed English hyphenation data.
12/// hypher = { version = "0.1", default-features = false, features = ["english"] }
13/// ```
14pub struct HYPHENATION;
15impl HYPHENATION {
16    /// Try to hyphenate the `word` using the `lang` rules.
17    ///
18    /// Returns a vector of indexes that allow a line break.
19    pub fn hyphenate(&self, lang: &Lang, word: &str) -> Vec<usize> {
20        self.hyphenate_opt(lang, word).unwrap_or_default()
21    }
22
23    /// Try to hyphenate the `word` using the `lang` rules.
24    ///
25    /// Returns a vector of indexes that allow a line break. Returns `None` if the `lang` is not supported or the
26    /// `word` contains non-word characters.
27    pub fn hyphenate_opt(&self, lang: &Lang, word: &str) -> Option<Vec<usize>> {
28        let lang = self.lang_to_hypher_lang(lang)?;
29        self.hyphenate_opt_lang(word, lang)
30    }
31
32    /// Get the best `hypher::Lang` for the `lang`.
33    pub fn lang_to_hypher_lang(&self, lang: &Lang) -> Option<hypher::Lang> {
34        hypher::Lang::from_iso(lang.0.language.as_str().as_bytes().try_into().ok()?)
35    }
36
37    /// Hyphenate with language already resolved.
38    ///
39    /// Returns `None` if the `word` contains non-word characters.
40    pub fn hyphenate_opt_lang(&self, word: &str, lang: hypher::Lang) -> Option<Vec<usize>> {
41        fn is_word(word: &str) -> bool {
42            // r"^\w+$"
43            !word.is_empty() && word.chars().all(|c| c.is_alphanumeric() || c == '_')
44        }
45        if !is_word(word) {
46            return None;
47        }
48
49        let ptr = word.as_ptr() as usize;
50        let breaks: Vec<_> = hypher::hyphenate(word, lang).skip(1).map(|s| s.as_ptr() as usize - ptr).collect();
51        Some(breaks)
52    }
53}
