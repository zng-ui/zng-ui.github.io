15pub fn translate(dir: &str, from: &str, to: &str, replace: bool, check: bool) {
16    let dir_path = Path::new(dir);
17    if !dir_path.exists() {
18        fatal!("dir `{dir}` does not exist")
19    }
20
21    let from = if from.is_empty() {
22        let name = dir_path.file_name().unwrap().to_string_lossy();
23        let name = &*name;
24        if name == "template" {
25            println_v!("from language set to `en`, default language for `template` folder");
26            LanguageIdentifier::from_bytes(b"en").unwrap()
27        } else {
28            let name = name.strip_suffix("-machine").unwrap_or(name);
29            match LanguageIdentifier::from_bytes(name.as_bytes()) {
30                Ok(l) => {
31                    println_v!("from language language set to `{l}`, derived from `{name}` folder");
32                    l
33                }
34                Err(e) => fatal!("cannot define from language, `{name}` is not a language, {e}\nset --translate-from to resolve"),
35            }
36        }
37    } else {
38        match LanguageIdentifier::from_bytes(from.as_bytes()) {
39            Ok(l) => l,
40            Err(e) => fatal!("cannot translate from `{from}`, {e}"),
41        }
42    };
43    let to = {
44        let mut t = vec![];
45        for l_str in to.split(',') {
46            let l_str = l_str.trim();
47            if l_str.is_empty() {
48                continue;
49            }
50            let l_str = l_str.strip_suffix("-machine").unwrap_or(l_str);
51            match LanguageIdentifier::from_bytes(l_str.as_bytes()) {
52                Ok(l) => t.push(l),
53                Err(e) => fatal!("cannot translate to `{l_str}`, {e}"),
54            }
55        }
56        t
57    };
58    if to.is_empty() {
59        fatal!("--translate-to requires at least one language")
60    }
61
62    let pattern = dir_path.join("**/*.ftl");
63    let mut files = vec![];
64    for entry in glob::glob(&pattern.display().to_string()).unwrap_or_else(|e| fatal!("cannot read `{dir}`, {e}")) {
65        let entry = entry.unwrap_or_else(|e| fatal!("cannot read `{dir}` entry, {e}"));
66        let relative_entry = entry.strip_prefix(dir_path).unwrap();
67        let file = match std::fs::read_to_string(&entry) {
68            Ok(f) => f,
69            Err(e) => fatal!("cannot read `{}`, {}", entry.display(), e),
70        };
71        let mut hasher = sha2::Sha256::new();
72        hasher.update(file.as_bytes());
73        let hash = format!("{:x}", base16ct::HexDisplay(&hasher.finalize()));
74        files.push((relative_entry.to_owned(), file, hash));
75    }
76    let files = files;
77
78    println_v!("collected {} source files", files.len());
79
80    let dir_path = dunce::canonicalize(dir_path).unwrap();
81    let dir_parent = dir_path.parent().unwrap();
82
83    let translator = Translator::default();
84    println_v!("using `{}`, RPM: {}", translator.path.display(), translator.rpm);
85
86    to.into_par_iter().for_each(|to| {
87        if to == from {
88            return;
89        }
90        println!("translating to {to}-machine");
91
92        let non_machine_dir = dir_parent.join(to.to_string());
93        if non_machine_dir.exists() {
94            return;
95        }
96
97        let dir_to = dir_parent.join(format!("{to}-machine"));
98
99        files.par_iter().for_each(|(relative_path, file, hash)| {
100            let file_to = dir_to.join(relative_path);
101            println_v!("  {}", file_to.display());
102
103            let _ = util::check_or_create_dir_all(check, file_to.parent().unwrap());
104
105            const HEADER_PREFIX: &str = "### Machine translated by `cargo zng l10n`, ";
106
107            let mut stale = false;
108            match fs::File::open(&file_to) {
109                Ok(f) => {
110                    if !replace {
111                        let mut f = io::BufReader::new(f);
112                        let mut header = String::new();
113                        let _ = f.read_line(&mut header);
114                        if let Some(h) = header.strip_prefix(HEADER_PREFIX) {
115                            if hash == h.trim_end() {
116                                // translation is up-to-date
117                                return;
118                            }
119                            stale = true;
120                        } else {
121                            // translation not generated by us, assume is up-to-date
122                            return;
123                        }
124                    }
125                }
126                Err(_) => {
127                    if check {
128                        fatal!("expected `{}`", file_to.display());
129                    }
130                }
131            }
132
133            if !check && (replace || stale || !file_to.exists()) {
134                let r = translator.translate(&from, &to, file);
135                let write = || -> io::Result<()> {
136                    let mut f = fs::File::create(&file_to)?;
137                    f.write_all(HEADER_PREFIX.as_bytes())?;
138                    f.write_all(hash.as_bytes())?;
139                    f.write_all("\n\n".as_bytes())?;
140                    f.write_all(r.as_bytes())?;
141                    Ok(())
142                };
143                if let Err(e) = write() {
144                    fatal!("cannot write `{}`, {}", file_to.display(), e);
145                }
146            }
147        });
148    });
149}

151struct Translator {
152    path: PathBuf,
153    rpm: u64,
154    limiter: LeakyBucket,
155}
156impl Default for Translator {
157    fn default() -> Self {
158        let install_dir = std::env::current_exe().unwrap();
159        let install_dir = install_dir.parent().unwrap();
160
161        // find translator
162        let mut t = if let Ok(translator) = std::env::var("ZNG_L10N_TRANSLATOR") {
163            let p = if translator.contains('/') || translator.contains('\\') {
164                PathBuf::from(&translator)
165            } else {
166                install_dir.join(format!("zng-l10n-translator-{translator}{}", std::env::consts::EXE_SUFFIX))
167            };
168            if !p.exists() {
169                fatal!("cannot find translator `{translator}`");
170            }
171            Translator {
172                path: p,
173                rpm: 0,
174                limiter: LeakyBucket::builder().build(),
175            }
176        } else {
177            let translators_pattern = install_dir.join("zng-l10n-translator-*").display().to_string();
178            let mut options = vec![];
179            for opt in glob::glob(&translators_pattern).unwrap() {
180                options.push(opt.unwrap());
181            }
182            if options.is_empty() {
183                fatal!("no translator installed\n   install a zng-l10n-translator-* crate\nor set ZNG_L10N_TRANSLATOR to a path")
184            } else if options.len() > 1 {
185                let mut names: Vec<_> = options
186                    .iter()
187                    .map(|p| {
188                        p.file_name()
189                            .unwrap()
190                            .to_string_lossy()
191                            .trim_start_matches("zng-l10n-translator-")
192                            .trim_end_matches(std::env::consts::EXE_SUFFIX)
193                            .to_owned()
194                    })
195                    .collect();
196                names.sort();
197                let names = names.join(", ");
198                fatal!("multiple translators installed\n    set ZNG_L10N_TRANSLATOR to one of: {names}");
199            }
200            Translator {
201                path: options.remove(0),
202                rpm: 0,
203                limiter: LeakyBucket::builder().build(),
204            }
205        };
206
207        // request limits
208        t.read_limits();
209
210        t
211    }
212}
213impl Translator {
214    fn read_limits(&mut self) {
215        #[derive(serde::Deserialize)]
216        struct Limits {
217            #[serde(rename = "requests-per-minute")]
218            rpm: u64,
219        }
220
221        let output = std::process::Command::new(&self.path).arg("--limits").output().unwrap();
222        if output.status.success() {
223            match serde_json::from_slice::<Limits>(&output.stdout) {
224                Ok(l) => {
225                    self.rpm = l.rpm;
226                    let t: u32 = self.rpm.try_into().unwrap();
227                    self.limiter = LeakyBucket::builder()
228                        .max(t)
229                        .tokens(t)
230                        .refill_amount(t)
231                        .refill_interval(Duration::from_mins(1))
232                        .build();
233                }
234                Err(e) => {
235                    let limits = String::from_utf8_lossy(&output.stdout);
236                    fatal!("invalid response to --limits, {e}\n{limits}")
237                }
238            }
239        } else {
240            let error = String::from_utf8_lossy(&output.stderr);
241            fatal!(
242                "{error}\n{} exited with code {}",
243                self.path.file_name().unwrap().to_string_lossy(),
244                output.status.code().unwrap_or(0)
245            );
246        }
247    }
248
249    fn translate(&self, from: &LanguageIdentifier, to: &LanguageIdentifier, file: &str) -> String {
250        self.limiter.acquire_one();
251
252        let mut cmd = std::process::Command::new(&self.path)
253            .arg("--from-lang")
254            .arg(from.to_string())
255            .arg("--to-lang")
256            .arg(to.to_string())
257            .stdin(std::process::Stdio::piped())
258            .stdout(std::process::Stdio::piped())
259            .stderr(std::process::Stdio::piped())
260            .spawn()
261            .unwrap();
262
263        let mut stdin = cmd.stdin.take().unwrap();
264        stdin.write_all(file.as_bytes()).unwrap();
265        stdin.flush().unwrap();
266        drop(stdin);
267
268        let s = cmd.wait().unwrap();
269        if !s.success() {
270            let mut error = String::new();
271            cmd.stderr.unwrap().read_to_string(&mut error).unwrap();
272            fatal!(
273                "{error}\n{} exited with code {}",
274                self.path.file_name().unwrap().to_string_lossy(),
275                s.code().unwrap_or(0)
276            );
277        }
278        let mut out = String::new();
279        cmd.stdout.unwrap().read_to_string(&mut out).unwrap();
280
281        if let Err(e) = fluent_syntax::parser::parse(out.as_str()) {
282            fatal!("translator output is not valid Fluent, {}", &e.1[0]);
283        }
284
285        out
286    }
