49/// Represents an install and uninstall task implementation.
50///
51/// Setup tasks runs in steps, steps do not necessarily run on the same process, communication
52/// between steps is done using serialized data. The steps are implemented as associated functions,
53/// not methods, the task type is not instantiated.
54///
55/// Each step runs for all tasks on the setup list, before moving to the next step.
56///
57/// # Install Steps
58///
59/// 1 - If the user did not cancel, [`SetupTask::prepare_install`] is called.
60/// 2.a - If did not cancel, [`SetupTask::install`] is called, the user cannot cancel once this starts.
61/// 2.b - If did cancel, [`SetupTask::cancel_install`] is called.
62///
63/// Note that steps 1 and 2 might not run on the same process. A case where this happens is a self-updater
64/// that starts preparing to install the update while it is still running.
65///
66/// # Uninstall Steps
67///
68/// 1 - [`SetupTask::Install`] data is deserialized from the install log.
69/// 2 - [`SetupTask::validate_uninstall`] is called.
70/// 3 - If did not cancel, [`SetupTask::uninstall`] is called.
71///
72/// # Register
73///
74/// Custom setup task types must be registered with [`SETUP.register_task_type`] otherwise install and uninstall
75/// will fail with [`SetupTaskError::UnknownType`].
76///
77/// [`SETUP.register_task_type`]: crate::SETUP::register_task_type
78pub trait SetupTask: Sized {
79    /// Install config type.
80    type InstallConfig: Any + Send;
81    /// Prepared install data type.
82    type PrepareInstall: ConfigValue;
83    /// Installed data type.
84    type Install: ConfigValue;
85
86    /// Unique ID for the task type.
87    fn task_type_id() -> TaskTypeId;
88
89    /// Run all expensive install operations that can run without affecting the system or previous installs.
90    ///
91    /// This step **must not** cause any change that affects existing install, even if reversible, it must only
92    /// run all potentially expensive tasks in such a way that the final *commit* can happen quickly.
93    ///
94    /// The user may cancel the install at any time, if possible monitor the [`cancel`] var and return
95    /// early on cancel. Implement cancellation cleanup on [`cancel_install`].
96    ///
97    /// [`cancel_install`]: Self::cancel_install
98    /// [`cancel`]: PrepareInstallArgs::cancel
99    fn prepare_install(
100        args: PrepareInstallArgs<Self>,
101    ) -> impl Future<Output = Result<Self::PrepareInstall, SetupTaskError>> + Send + 'static;
102
103    /// Commit prepared install changes.
104    ///
105    /// The user cannot cancel installation when this step is running. Progress indicators will only show *indeterminate*
106    /// with the expectation this step will finish quickly.
107    ///
108    /// Install must not fail at the first error encountered, a best attempt to apply all install steps must be made,
109    /// errors can be aggregated on the [`InstallTaskError::error`]. The [`InstallTaskError::clean_data`] must include uninstall instructions
110    /// for all successful steps, best attempt of partial steps and any data from the previous version that was not replaced in case
111    /// it is installing an update.
112    ///
113    /// [`prepare_install`]: Self::prepare_install
114    fn install(args: InstallArgs<Self>) -> impl Future<Output = Result<Self::Install, InstallTaskError<Self::Install>>> + Send + 'static;
115
116    /// Cancel prepared install changes.
117    ///
118    /// This is called if the user requested cancel during or after [`prepare_install`] and before [`install`].
119    ///
120    /// This step must find and cleanup all prepared changes, such as temporary files. The cancel logic must be resilient to
121    /// partial changes as [`prepare_install`] might return early due to user cancel or an error.
122    ///
123    /// [`prepare_install`]: Self::prepare_install
124    /// [`install`]: Self::install
125    fn cancel_install(args: CancelInstallArgs<Self>) -> impl Future<Output = Result<(), SetupTaskError>> + Send + 'static;
126
127    /// Validate the install state for uninstall.
128    ///
129    /// This step **must not** make any changes to the file system, not even creating temp files. This step
130    /// allows tasks to validate the install state before [`uninstall`] makes irreversible changes.
131    ///
132    /// This step is not expected to take long, but if it does check the [`cancel`] flag to avoid unnecessary work.
133    /// If the uninstall is canceled when another task is preparing after this one the returned data is just dropped.
134    ///
135    /// This step returns a validation error or the corrected install data.
136    ///
137    /// [`uninstall`]: Self::uninstall
138    /// [`cancel`]: ValidateUninstallArgs::cancel
139    fn validate_uninstall(
140        args: ValidateUninstallArgs<Self>,
141    ) -> impl Future<Output = Result<Self::Install, SetupTaskError>> + Send + 'static;
142
143    /// Uninstall.
144    ///
145    /// The user cannot cancel uninstallation when this step is running.
146    ///
147    /// Uninstall is idempotent, it must not fail in case a step is already completed, for example, if the task must remove a file
148    /// and it is not found, that is not an error. Task runners can retry partially run uninstall with an install data clone.
149    ///
150    /// Uninstall must not fail at the first error encountered, a best attempt to apply all uninstall steps must be made,
151    /// errors can be aggregated on the [`SetupTaskError`].
152    fn uninstall(args: UninstallArgs<Self>) -> impl Future<Output = Result<(), SetupTaskError>> + Send + 'static;
153}

155/// Arguments for [`SetupTask::prepare_install`]
156#[non_exhaustive]
157pub struct PrepareInstallArgs<T: SetupTask> {
158    /// Config for the new installation.
159    pub config: T::InstallConfig,
160
161    /// Data from the previous installation that is being replaced with this one.
162    ///
163    /// This is set if is installing over a previous installation and the same task is present
164    /// on the new installation.
165    pub update: Option<T::Install>,
166
167    /// Progress indicator for the task. Starts as [`Progress::indeterminate`] by default.
168    pub progress: Var<Progress>,
169    /// Read-only var that is `true` if the user cancels the installation.
170    ///
171    /// If possible check this flag often and return immediately on cancel. The *prepare install*
172    /// step is not expected to cleanup on cancel, just return immediately.
173    pub cancel: Var<bool>,
174}

221/// Represents a [`SetupTask`] step error.
222///
223/// Some tasks may continue after an error in a best attempt to at least complete most of the work,
224/// this can cause multiple errors to aggregate. In these cases the [`Error::source`] is the first
225/// error.
226#[derive(Debug, Clone)]
227#[non_exhaustive]
228pub enum SetupTaskError {
229    /// Task data is in an unexpected format.
230    CorruptedTaskData(Arc<dyn Error + Send + Sync>),
231    /// Task type is not [registered].
232    ///
233    /// [registered]: crate::SETUP::register_task_type
234    UnknownType(TaskTypeId),
235    /// IO errors associated with a file or directory path.
236    Io(Vec<(PathBuf, Arc<std::io::Error>)>),
237    /// Other errors.
238    Other(Vec<Arc<dyn Error + Send + Sync>>),
239}
240impl SetupTaskError {
241    /// New `Io` error with a single entry.
242    pub fn io(related_path: PathBuf, error: std::io::Error) -> Self {
243        Self::Io(vec![(related_path, Arc::new(error))])
244    }
245
246    /// New `Other` error with a single entry.
247    pub fn other(error: impl Error + Send + Sync + 'static) -> Self {
248        Self::Other(vec![Arc::new(error)])
249    }
250}
251/// Inner errors only compare `Arc` pointer.
252impl PartialEq for SetupTaskError {
253    fn eq(&self, other: &Self) -> bool {
254        match (self, other) {
255            (Self::CorruptedTaskData(a), Self::CorruptedTaskData(b)) => Arc::ptr_eq(a, b),
256            (Self::UnknownType(a), Self::UnknownType(b)) => a == b,
257            (Self::Io(a), Self::Io(b)) => a.len() == b.len() && a.iter().zip(b).all(|(a, b)| Arc::ptr_eq(&a.1, &b.1) && a.0 == b.0),
258            (Self::Other(a), Self::Other(b)) => a.len() == b.len() && a.iter().zip(b).all(|(a, b)| Arc::ptr_eq(a, b)),
259            _ => false,
260        }
261    }
262}
263impl fmt::Display for SetupTaskError {
264    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
265        match self {
266            SetupTaskError::CorruptedTaskData(e) => write!(f, "corrupted task data, {e}"),
267            SetupTaskError::UnknownType(t) => write!(f, "unknown task type {t}"),
268            SetupTaskError::Io(e) => {
269                let tab = if e.len() > 1 { "   " } else { "" };
270                let mut sep = "";
271                if e.len() > 1 {
272                    write!(f, "{} io errors:", e.len())?;
273                    sep = "\n";
274                }
275                for (p, e) in e.iter() {
276                    write!(f, "{sep}{tab}{e}\n{tab}   related path: {}", p.display())?;
277                    sep = "\n";
278                }
279                if e.is_empty() { write!(f, "unknown io error") } else { Ok(()) }
280            }
281            SetupTaskError::Other(e) => {
282                let tab = if e.len() > 1 { "   " } else { "" };
283                let mut sep = "";
284                if e.len() > 1 {
285                    write!(f, "{} errors:", e.len())?;
286                    sep = "\n";
287                }
288                for e in e.iter() {
289                    write!(f, "{sep}{tab}{e}")?;
290                }
291                if e.is_empty() { write!(f, "unknown error") } else { Ok(()) }
292            }
293        }
294    }
295}
296impl Error for SetupTaskError {
297    fn source(&self) -> Option<&(dyn Error + 'static)> {
298        match self {
299            SetupTaskError::CorruptedTaskData(e) => Some(&**e),
300            Self::UnknownType(_) => None,
301            Self::Io(e) => Some(&e.first()?.1),
302            SetupTaskError::Other(e) => Some(&**e.first()?),
303        }
304    }
305}

307/// Error in a [`SetupTask::install`] task run.
308pub struct InstallTaskError<I> {
309    /// The error.
310    pub error: SetupTaskError,
311    /// Cleanup [`SetupTask::Install`] data.
312    ///
313    /// This must contain data to uninstall the partial committed changes, if there where any. Task runners may
314    /// use this to attempt a [`SetupTask::uninstall`] to cleanup the corrupted install.
315    ///
316    /// In case the install is an [update], this must also contain all data from the previous install that has not
317    /// been invalidated by the failed install.
318    ///
319    /// If this is `None` the task runner will show that the task corrupted the install and the changes made
320    /// cannot even be uninstalled. It will also assume that the [`SetupTask::PrepareInstall`] data was not
321    /// fully cleaned before the error.
322    ///
323    /// If this is `Some` the task runner will assume the [`SetupTask::PrepareInstall`] is fully cleaned. The
324    /// task must attempt to run a *cancel* on the partial prepared data that has not committed yet, if the
325    /// error was encountered before any changes where actually committed and all prepared changes where successfully
326    /// canceled this must be set to `Some` value that represents an *empty install* that the uninstall task will
327    /// recognize and immediately return success for.
328    ///
329    /// [update]: PrepareInstallArgs::update
330    pub clean_data: Option<I>,
331}
332impl<I> fmt::Debug for InstallTaskError<I> {
333    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
334        f.debug_struct("InstallTaskError")
335            .field("error", &self.error)
336            .field("clean_data.is_some()", &self.clean_data.is_some())
337            .finish()
338    }
339}
340impl<I> fmt::Display for InstallTaskError<I> {
341    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
342        fmt::Display::fmt(&self.error, f)
343    }
344}
345impl<I> std::error::Error for InstallTaskError<I> {
346    fn source(&self) -> Option<&(dyn Error + 'static)> {
347        Some(&self.error)
348    }
349}

351type BoxFutResult<T, E> = Pin<Box<dyn Future<Output = Result<T, E>> + Send + 'static>>;
352
353fn value_de<T: ConfigValue>(raw: RawConfigValue) -> Result<T, SetupTaskError> {
354    match raw.deserialize() {
355        Ok(r) => Ok(r),
356        Err(e) => Err(SetupTaskError::CorruptedTaskData(Arc::new(e))),
357    }
358}

360#[derive(Clone)]
361pub(crate) struct SetupTaskType {
362    pub task_type_id: fn() -> TaskTypeId,
363    #[allow(clippy::type_complexity)]
364    pub prepare_install:
365        fn(Box<dyn Any + Send>, Option<RawConfigValue>, Var<Progress>, Var<bool>) -> BoxFutResult<RawConfigValue, SetupTaskError>,
366    pub install: fn(RawConfigValue, Var<Progress>) -> BoxFutResult<RawConfigValue, InstallTaskError<RawConfigValue>>,
367    pub cancel_install: fn(RawConfigValue, Var<Progress>) -> BoxFutResult<(), SetupTaskError>,
368    pub validate_uninstall: fn(RawConfigValue, Var<Progress>, Var<bool>) -> BoxFutResult<RawConfigValue, SetupTaskError>,
369    pub uninstall: fn(RawConfigValue, Var<Progress>) -> BoxFutResult<(), SetupTaskError>,
370}
371impl SetupTaskType {
372    /// New task instance.
373    pub fn new<T: SetupTask>() -> Self {
374        Self {
375            task_type_id: T::task_type_id,
376            prepare_install: Self::raw_prepare_install::<T>,
377            install: Self::raw_install::<T>,
378            cancel_install: Self::raw_cancel_install::<T>,
379            validate_uninstall: Self::raw_validate_uninstall::<T>,
380            uninstall: Self::raw_uninstall::<T>,
381        }
382    }
383    fn raw_prepare_install<T: SetupTask>(
384        config: Box<dyn Any + Send>,
385        update: Option<RawConfigValue>,
386        progress: Var<Progress>,
387        cancel: Var<bool>,
388    ) -> BoxFutResult<RawConfigValue, SetupTaskError> {
389        Box::pin(async move {
390            let args = PrepareInstallArgs {
391                config: *config.downcast().unwrap(),
392                update: match update {
393                    Some(d) => value_de(d)?,
394                    None => None,
395                },
396                progress,
397                cancel,
398            };
399            let r = T::prepare_install(args).await?;
400            Ok(RawConfigValue::serialize(r).unwrap())
401        })
402    }
403    fn raw_install<T: SetupTask>(
404        data: RawConfigValue,
405        progress: Var<Progress>,
406    ) -> BoxFutResult<RawConfigValue, InstallTaskError<RawConfigValue>> {
407        Box::pin(async move {
408            let args = InstallArgs {
409                data: match value_de(data) {
410                    Ok(d) => d,
411                    Err(e) => {
412                        return Err(InstallTaskError {
413                            error: e,
414                            // can't cancel either without the data
415                            clean_data: None,
416                        });
417                    }
418                },
419                progress,
420            };
421            match T::install(args).await {
422                Ok(r) => Ok(RawConfigValue::serialize(r).unwrap()),
423                Err(e) => Err(InstallTaskError {
424                    error: e.error,
425                    clean_data: e.clean_data.map(|d| RawConfigValue::serialize(d).unwrap()),
426                }),
427            }
428        })
429    }
430    fn raw_cancel_install<T: SetupTask>(data: RawConfigValue, progress: Var<Progress>) -> BoxFutResult<(), SetupTaskError> {
431        Box::pin(async move {
432            let args = CancelInstallArgs {
433                data: value_de(data)?,
434                progress,
435            };
436            T::cancel_install(args).await
437        })
438    }
439    fn raw_validate_uninstall<T: SetupTask>(
440        data: RawConfigValue,
441        progress: Var<Progress>,
442        cancel: Var<bool>,
443    ) -> BoxFutResult<RawConfigValue, SetupTaskError> {
444        Box::pin(async move {
445            let args = ValidateUninstallArgs {
446                data: value_de(data)?,
447                progress,
448                cancel,
449            };
450            let r = T::validate_uninstall(args).await?;
451            Ok(RawConfigValue::serialize(r).unwrap())
452        })
453    }
454    fn raw_uninstall<T: SetupTask>(data: RawConfigValue, progress: Var<Progress>) -> BoxFutResult<(), SetupTaskError> {
455        Box::pin(async move {
456            let args = UninstallArgs {
457                data: value_de(data)?,
458                progress,
459            };
460            T::uninstall(args).await
461        })
462    }
463}

465#[allow(unused)]
466pub(crate) fn path_utf8(p: PathBuf) -> Result<String, SetupTaskError> {
467    match p.to_str() {
468        Some(s) => Ok(if cfg!(windows) {
469            s.replace('/', "\\")
470        } else {
471            s.replace('\\', "/")
472        }),
473        None => Err(SetupTaskError::io(
474            p,
475            io::Error::new(io::ErrorKind::InvalidData, "path must be utf-8"),
476        )),
477    }
478}
